cmake_minimum_required(VERSION 3.22)
project(ailocal LANGUAGES C CXX)
# The installable development APK must run the same optimized CPU kernels as a
# release. AGP's Debug default (-O0) makes multi-billion-parameter inference crawl.
if(ANDROID)
    add_compile_options(-O3)
    add_link_options("-Wl,-z,max-page-size=16384")
endif()
include(FetchContent)
set(AUREN_LIBRARY_OUTPUT "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
    # Official runtime-dispatched Android NEON, dot-product, FP16 and i8mm
    # variants. Never force a modern ISA on an older phone.
    set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(llama URL https://github.com/ggml-org/llama.cpp/archive/9581154a2250cc9e6948e4c2a2fc4b4e19d1a57e.tar.gz)
FetchContent_MakeAvailable(llama)
# Upstream places plugin libraries under bin/. Put every shared dependency and
# CPU module in AGP's native output directory so the APK contains the full engine.
function(auren_native_outputs directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        get_target_property(kind "${target}" TYPE)
        if(kind STREQUAL "SHARED_LIBRARY" OR kind STREQUAL "MODULE_LIBRARY")
            set_target_properties("${target}" PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${AUREN_LIBRARY_OUTPUT}")
        endif()
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        auren_native_outputs("${child}")
    endforeach()
endfunction()
if(AUREN_LIBRARY_OUTPUT)
    auren_native_outputs("${CMAKE_CURRENT_SOURCE_DIR}")
endif()
add_library(ailocal SHARED engine.cpp)
target_compile_features(ailocal PRIVATE cxx_std_17)
target_link_libraries(ailocal PRIVATE llama mtmd android log)
target_link_options(ailocal PRIVATE "-Wl,-z,max-page-size=16384")
